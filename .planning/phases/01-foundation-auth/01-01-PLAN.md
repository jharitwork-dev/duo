1. Create `drizzle.config.ts` at project root:
```typescript
import 'dotenv/config';
import { defineConfig } from 'drizzle-kit';

export default defineConfig({
  schema: './src/db/schema',
  out: './drizzle',
  dialect: 'postgresql',
  dbCredentials: {
    url: process.env.DATABASE_URL_UNPOOLED!,
  },
});
```

2. Create `src/db/index.ts` — Drizzle client with Neon WebSocket Pool (per D-08):
```typescript
import { drizzle } from 'drizzle-orm/neon-serverless';
import { Pool, neonConfig } from '@neondatabase/serverless';
import ws from 'ws';
import * as schema from './schema';

// WebSocket polyfill for Node.js (not needed in edge/CF Workers)
neonConfig.webSocketConstructor = ws;

const pool = new Pool({ connectionString: process.env.DATABASE_URL! });

export const db = drizzle(pool, { schema });
```

3. Create schema files. All tables use snake_case plural names. All PKs are `text` with cuid2 `$defaultFn`. All user FKs are `text` referencing Clerk user IDs directly (per D-07). All tables have `created_at` timestamp defaultNow.

`src/db/schema/classrooms.ts`:
- `classrooms` table: id (text PK, cuid2), name (text notNull), description (text nullable), created_by (text notNull — Clerk userId), created_at (timestamp defaultNow notNull), updated_at (timestamp defaultNow notNull)
- `classroom_members` table: id (text PK, cuid2), classroom_id (text notNull, FK -> classrooms.id onDelete cascade), user_id (text notNull — Clerk userId), role (text enum ['teacher', 'student'] notNull), joined_at (timestamp defaultNow notNull). Unique constraint on (classroom_id, user_id).

`src/db/schema/groups.ts`:
- `groups` table: id (text PK, cuid2), classroom_id (text notNull, FK -> classrooms.id onDelete cascade), name (text notNull), created_by (text notNull), created_at (timestamp defaultNow notNull), updated_at (timestamp defaultNow notNull)
- `group_members` table: id (text PK, cuid2), group_id (text notNull, FK -> groups.id onDelete cascade), user_id (text notNull — Clerk userId), joined_at (timestamp defaultNow notNull). Unique constraint on (group_id, user_id).

`src/db/schema/phases.ts`:
- `phases` table: id (text PK, cuid2), group_id (text notNull, FK -> groups.id onDelete cascade), name (text notNull), description (text nullable), order_index (integer notNull default 0), status (text enum ['locked', 'active', 'completed'] notNull default 'locked'), is_free_access (boolean notNull default false), deadline (timestamp nullable), created_by (text notNull), created_at (timestamp defaultNow notNull), updated_at (timestamp defaultNow notNull)

`src/db/schema/todos.ts`:
- `todos` table: id (text PK, cuid2), phase_id (text notNull, FK -> phases.id onDelete cascade), title (text notNull), description (text nullable), notes (text nullable — teacher notes, rich text JSON stored as text), order_index (integer notNull default 0), submission_mode (text enum ['group', 'individual'] notNull default 'group'), deadline (timestamp nullable), created_by (text notNull), created_at (timestamp defaultNow notNull), updated_at (timestamp defaultNow notNull)
- `todo_attachments` table: id (text PK, cuid2), todo_id (text notNull, FK -> todos.id onDelete cascade), file_name (text notNull), file_key (text notNull — R2 object key), content_type (text notNull), file_size (integer notNull), uploaded_by (text notNull), created_at (timestamp defaultNow notNull)

`src/db/schema/submissions.ts`:
- `submissions` table: id (text PK, cuid2), todo_id (text notNull, FK -> todos.id onDelete cascade), submitted_by (text notNull — Clerk userId), group_id (text nullable, FK -> groups.id onDelete set null — set when submission_mode is 'group'), status (text enum ['pending', 'approved', 'rejected'] notNull default 'pending'), text_content (text nullable — for text submissions), link_url (text nullable — for link submissions), reviewed_by (text nullable), reviewed_at (timestamp nullable), created_at (timestamp defaultNow notNull), updated_at (timestamp defaultNow notNull)
- `submission_files` table: id (text PK, cuid2), submission_id (text notNull, FK -> submissions.id onDelete cascade), file_name (text notNull), file_key (text notNull — R2 object key), content_type (text notNull), file_size (integer notNull), created_at (timestamp defaultNow notNull)

`src/db/schema/comments.ts`:
- `comments` table: id (text PK, cuid2), submission_id (text notNull, FK -> submissions.id onDelete cascade), user_id (text notNull — Clerk userId), content (text notNull), created_at (timestamp defaultNow notNull), updated_at (timestamp defaultNow notNull)
